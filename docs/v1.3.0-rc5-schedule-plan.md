`plan.date = today + 1` (life_simulator.py:436) + 双路径 `_last_plan_date` 语义不一致:
- `main.py:938` (cron): `_last_plan_date = today_str`

### 1. 新文件 `emotion_spirit/utils/time_utils.py`

```python
"""时间工具: 逻辑日 (6am 边界) + 日程显示标签. v1.3.0 rc.5.

用户模型: day(n) 的 6am 前 = day(n-1). cron 在 02:00 (6am 前) 触发,
生成即将到来的逻辑日的 plan. logical_today() 用于显示标签, 不用于 plan.date
(plan.date = date.today() at 02:00 = 即将到来的逻辑日, 二者天然对齐).
"""
from __future__ import annotations
from datetime import date, datetime, timedelta


def logical_today(now: datetime | None = None) -> date:
    """当前逻辑日 (6am 为界, 6am 前 = 昨天).

    Args:
        now: 当前时间 (默认 datetime.now()). 测试可注入.

    Returns:
        逻辑日 date. 06:00 前 = 日历昨天, 06:00 后 = 日历今天.
    """
    from emotion_spirit.core.config import LIFE_SIM_V2_CONFIG
    start_hour = LIFE_SIM_V2_CONFIG.get("logical_day_start_hour", 6)
    now = now or datetime.now()
    today = now.date()
    if now.hour < start_hour:
        return today - timedelta(days=1)
    return today


def plan_date_label(date_str: str, now: datetime | None = None) -> str:
    """日程显示标签, 基于逻辑日边界.

    plan.date 是日历日 (生成时 date.today()). 标签按 plan.date 与 logical_today
    的差值显示: 0 = 今天, 1 = 明天, -1 = 昨天, 其他 = 日期.

    Args:
        date_str: plan.date (ISO 格式, e.g. "2026-07-05").
        now: 当前时间 (测试注入).

    Returns:
        "今天日程 (2026-07-05)" / "明天日程 (2026-07-06)" / "日程计划 (2026-07-07)"
    """
    plan_date = date.fromisoformat(date_str)
    today = logical_today(now)
    delta = (plan_date - today).days
    if delta == 0:
        return f"今天日程 ({date_str})"
    if delta == 1:
        return f"明天日程 ({date_str})"
    if delta == -1:
        return f"昨天日程 ({date_str})"
    return f"日程计划 ({date_str})"
```

### 3. `main.py` — 重构 cron loop + 加 catch-up + 统一 dedup

**替换整个 `_schedule_plan_generation_loop` 方法 (行 902-996)** 为下面两个方法. 保留所有逻辑 (suppression L2 回写 + 梦境生成 + save_if_dirty), 只是抽出 `_maybe_generate_plan` 供 catch-up + cron 复用:

```python
    async def _schedule_plan_generation_loop(self) -> None:
        """每天 plan_generate_hour (默认 2am, < 6am 逻辑日边界) 生成当日日程.

        v1.3.0 rc.5 Bug-I:
        - plan.date = date.today() (不再是 today+1)
        - _last_plan_date = plan.date (与 commands.py setup_init 路径统一, 防 dedup 双路径不一致)
        - 启动 catch-up: bot 重启漏 02:00 时补生成当天 plan
        """
        from emotion_spirit.core.config import LIFE_SIM_V2_CONFIG

        # 启动 catch-up: bot 重启漏掉 02:00 时, 立即补生成当天 plan
        await self._maybe_generate_plan()

        while True:
            try:
                now = datetime.now()
                target_hour = LIFE_SIM_V2_CONFIG.get("plan_generate_hour", 2)
                target = now.replace(hour=target_hour, minute=0, second=0, microsecond=0)
                if now >= target:
                    target += timedelta(days=1)
                wait_seconds = (target - now).total_seconds()
                logger.info("emotion_spirit: 日程生成定时器，下次触发 %s (%.0f 秒后)", target, wait_seconds)
                await asyncio.sleep(wait_seconds)
                await self._maybe_generate_plan()
            except asyncio.CancelledError:
                break
            except Exception:
                logger.warning("emotion_spirit: 日程生成循环异常", exc_info=True)
                await asyncio.sleep(60)

    async def _maybe_generate_plan(self) -> None:
        """生成当天 plan (dedup: _last_plan_date == today 则跳过). rc.5 抽出供 catch-up + cron 复用."""
        # _life_sim_v2 未就绪 (init race) 则跳过, 等 02:00 cron 再试
        if not hasattr(self, '_life_sim_v2') or self._life_sim_v2 is None:
            logger.debug("emotion_spirit: _life_sim_v2 未就绪, 跳过 plan 生成")
            return

        today_str = date.today().isoformat()
        if self._last_plan_date == today_str:
            logger.info("emotion_spirit: 今天已生成日程，跳过")
            return

        # 生成日程
        personality = self._get_current_personality_dict()
        recent_memories = self._get_recent_memory_texts(limit=5)
        yesterday_events = self._get_yesterday_events()

        plan = await self._life_sim_v2.generate_daily_plan(
            personality=personality,
            recent_memories=recent_memories,
            yesterday_events=yesterday_events,
            user_activity=(
                self._latest_user_activity
                if hasattr(self, '_latest_user_activity') else None
            ),
        )
        # rc.5 Bug-I: 用 plan.date (与 commands.py:162 setup_init 路径统一).
        # plan.date = today (life_simulator 方案 A), 与 today_str 等价, 但语义统一防双路径不一致.
        self._last_plan_date = plan.date

        # v1.2.9 HP-3: suppression L2 定期回写 (每天 1 次, 慢变量)
        try:
            defense_states = self._defense_modulator.compute_defense_states(
                personality=personality,
                signals=None,
                body_state=self._body_state.default() if hasattr(self, "_body_state") else None,
                intimacy_level=0.5,
                context={},
                force_state=(
                    self._force_dynamics.force_state_from_labels(self._labels)
                    if hasattr(self, "_force_dynamics") and hasattr(self, "_labels")
                    else None
                ),
                conscience_pressure=self._conscience.get_pressure() if hasattr(self, "_conscience") else 0.0,
            )
            self._defense_modulator.apply_event("suppression", intensity=defense_states.suppression_level)
            logger.debug("emotion_spirit: suppression L2 回写 level=%.3f", defense_states.suppression_level)
        except Exception:
            logger.debug("emotion_spirit: suppression L2 回写失败", exc_info=True)

        logger.info(
            "emotion_spirit: 日程已生成 %s, %d 个事件",
            plan.date, len(plan.events),
        )

        # v1.1.0B: 深度睡眠梦境生成
        if hasattr(self, '_dream_generator') and self._dream_generator._llm:
            try:
                sleep_hours = 6.0
                rounds = self._dream_generator.compute_dream_rounds(sleep_hours, personality)
                dream_seed = plan.dream_seed or ""
                recent_events = [e.activity for e in plan.events if e.status == "done"]
                dream = await self._dream_generator.generate_deep_sleep_dream(
                    personality=personality,
                    dream_seed=dream_seed,
                    recent_events=recent_events,
                )
                if dream:
                    self._pool.add(
                        text=f"[梦境] {dream[:200]}",
                        raw_weight=0.4,
                        phi=0.2,
                        tags=["dream", "deep_sleep"],
                        source_user="dream_generator",
                    )
                    logger.info("emotion_spirit: 深度睡眠梦境已生成 (%d 轮)", rounds)
            except Exception:
                logger.debug("emotion_spirit: 深度睡眠梦境生成失败", exc_info=True)

        self._save_if_dirty()
```

### 6. 新测试 `tests/test_schedule_plan_loop_cron_consistency.py`

```python
"""v1.3.0 rc.5 Bug-I: schedule cron 一致性 + 6am 逻辑日边界测试.

验证:
- plan.date = date.today() (不是 today+1)
- _last_plan_date 与 plan.date 统一 (双路径一致)
- cron 跨日重生成 (dedup 不自锁)
- logical_today 6am 边界
- plan_date_label 今天/明天
- catch-up: _last_plan_date != today 时 _maybe_generate_plan 生成
"""
import asyncio
from datetime import date, datetime, timedelta
from unittest.mock import AsyncMock, MagicMock, patch


def test_logical_today_6am_boundary():
    """6am 前 = 昨天, 6am 后 = 今天."""
    from emotion_spirit.utils.time_utils import logical_today
    # 03:00 (before 6am) → yesterday
    assert logical_today(datetime(2026, 7, 5, 3, 0)) == date(2026, 7, 4)
    # 06:00 (at boundary) → today
    assert logical_today(datetime(2026, 7, 5, 6, 0)) == date(2026, 7, 5)
    # 14:00 (after 6am) → today
    assert logical_today(datetime(2026, 7, 5, 14, 0)) == date(2026, 7, 5)
    # 23:59 → today
    assert logical_today(datetime(2026, 7, 5, 23, 59)) == date(2026, 7, 5)


def test_plan_date_label_today_tomorrow():
    """plan_date_label: 14:00 plan.date=today → 今天; 03:00 plan.date=today → 明天."""
    from emotion_spirit.utils.time_utils import plan_date_label
    # 14:00, plan.date = today (2026-07-05) → 今天日程
    assert plan_date_label("2026-07-05", datetime(2026, 7, 5, 14, 0)) == "今天日程 (2026-07-05)"
    # 03:00 (before 6am), plan.date = 2026-07-05 (generated at 02:00) → 明天日程
    assert plan_date_label("2026-07-05", datetime(2026, 7, 5, 3, 0)) == "明天日程 (2026-07-05)"
    # 01:00 next day (before 6am, still living day D), plan.date = D → 今天日程
    assert plan_date_label("2026-07-05", datetime(2026, 7, 6, 1, 0)) == "今天日程 (2026-07-05)"


def test_plan_date_is_today_not_tomorrow():
    """Bug-I 核心: generate_daily_plan 设 plan.date = today (不是 today+1)."""
    from emotion_spirit.regulation.life_simulator import LifeSimulatorV2
    fixed_today = date(2026, 7, 5)
    sim = LifeSimulatorV2(llm=None)
    with patch("emotion_spirit.regulation.life_simulator._dt") as dt_mod, \
         patch("emotion_spirit.regulation.life_simulator.time.time", return_value=1000.0):
        dt_mod.date.today.return_value = fixed_today
        dt_mod.timedelta = timedelta
        # generate_plan_template 走模板路径 (不需 LLM)
        plan = asyncio.get_event_loop().run_until_complete(
            sim.generate_daily_plan(personality={"warmth_bias": 0.5}, recent_memories=[], yesterday_events=[])
        )
    assert plan.date == "2026-07-05", f"plan.date 应为今天 2026-07-05, 实际 {plan.date}"


def test_maybe_generate_plan_dedup_and_unified_last_plan_date():
    """_maybe_generate_plan: dedup (_last==today 跳过) + _last_plan_date = plan.date (统一)."""
    from main import EmotionSpiritPlugin

def test_cron_regenerates_across_day_rollover():
    """Bug-I: 跨日后 cron 应重新生成 (dedup 不自锁)."""
    from main import EmotionSpiritPlugin
    plugin = EmotionSpiritPlugin.__new__(EmotionSpiritPlugin)
    plugin._life_sim_v2 = MagicMock()
    plan1, plan2 = MagicMock(), MagicMock()
    plan1.date = "2026-07-05"; plan1.events = []; plan1.dream_seed = ""
    plan2.date = "2026-07-06"; plan2.events = []; plan2.dream_seed = ""
    plugin._life_sim_v2.generate_daily_plan = AsyncMock(side_effect=[plan1, plan2])
    plugin._get_current_personality_dict = MagicMock(return_value={})
    plugin._get_recent_memory_texts = MagicMock(return_value=[])
    plugin._get_yesterday_events = MagicMock(return_value=[])
    plugin._save_if_dirty = MagicMock()

    # Day 1: 7/5, _last="" → 生成, _last=plan1.date=7/5

    assert plugin._last_plan_date == "2026-07-06", f"跨日应重新生成, _last={plugin._last_plan_date!r}"
    assert plugin._life_sim_v2.generate_daily_plan.call_count == 2
```
